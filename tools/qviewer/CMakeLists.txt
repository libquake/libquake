cmake_minimum_required(VERSION 3.10)

project(
  qviewer
  VERSION 1.0.0
  LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/tools)

# Dependencies
set(RAYLIB_VERSION 5.5)
find_package(raylib ${RAYLIB_VERSION} QUIET) # QUIET or REQUIRED

if(NOT raylib_FOUND) # If there's none, fetch and build raylib
  include(FetchContent)
  FetchContent_Declare(
    raylib
    DOWNLOAD_EXTRACT_TIMESTAMP OFF
    URL https://github.com/raysan5/raylib/archive/refs/tags/${RAYLIB_VERSION}.tar.gz
  )
  FetchContent_GetProperties(raylib)

  if(NOT raylib_POPULATED) # Have we downloaded raylib yet?
    set(FETCHCONTENT_QUIET NO)
    FetchContent_Populate(raylib)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE) # don't build the supplied examples
    add_subdirectory(${raylib_SOURCE_DIR} ${raylib_BINARY_DIR})
  endif()
endif()

# The shaders are plain GLSL files in shaders/. They are embedded into the binary at configure time, and qviewer
# can reload them from the source folder while it runs (F5), so they can be edited without rebuilding.
set(QVIEWER_SHADERS world.vert world.frag sky.frag bloom_extract.frag bloom_blur.frag bloom_composite.frag)
set(EMBEDDED_SHADERS "#pragma once\n\n// generated by CMake from tools/qviewer/shaders, do not edit\n\n")
string(APPEND EMBEDDED_SHADERS "struct EmbeddedShader {\n  const char *name;\n  const char *code;\n};\n\n")
string(APPEND EMBEDDED_SHADERS "inline const EmbeddedShader EMBEDDED_SHADERS[] = {\n")
foreach(SHADER ${QVIEWER_SHADERS})
  set(SHADER_PATH ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER})
  file(READ ${SHADER_PATH} SHADER_CODE)
  string(APPEND EMBEDDED_SHADERS "    {\"${SHADER}\", R\"glsl(${SHADER_CODE})glsl\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SHADER_PATH})
endforeach()
string(APPEND EMBEDDED_SHADERS "};\n")
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_shaders.h CONTENT "${EMBEDDED_SHADERS}")

add_executable(${PROJECT_NAME}
  main.cpp
  viewer.cpp
  fly_camera.cpp
  level_models.cpp
  level_textures.cpp
  lightmap_animation.cpp
  renderer/bloom.cpp
  renderer/shader_library.cpp
  renderer/world_renderer.cpp
)

target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_definitions(${PROJECT_NAME} PRIVATE QVIEWER_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")

target_link_libraries(
  ${PROJECT_NAME}
  ${OPENGL_LIBRARY}
  raylib
  quakelib
)

target_include_directories(${PROJECT_NAME} PRIVATE "../include/")
